cmake_minimum_required(VERSION 3.16)
project(embedded_core LANGUAGES CXX)
enable_testing()
add_executable(embedded_core_test embedded_core.cpp)
target_compile_features(embedded_core_test PRIVATE cxx_std_11)
target_compile_definitions(embedded_core_test PRIVATE EMBEDDED_TEST=1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(embedded_core_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME embedded_core_test COMMAND embedded_core_test)
